% Enhanced resume subheading with better typography and spacing
\newcommand{\resumeSubheading}[4]{
  \vspace{-3pt} \item 
    \begin{tabular*}{\dimexpr\textwidth-0.3in\relax}[t]{l@{\extracolsep{\fill}}r}
      \textbf{\color{accentblue}#1} & \color{mediumgray}#2 \\
      \color{mediumgray}#3 & \color{mediumgray}#4 \\
    \end{tabular*}
  \vspace{-9pt}
}

\newcommand{\educationItem}[4]{
  {\dimexpr\textwidth-0.3in\relax}{\extracolsep{\fill}}
      \textbf{\color{accentblue}#1} & \color{mediumgray}#2 \\
      \color{mediumgray}#3 & \color{mediumgray}#4 \\
}
