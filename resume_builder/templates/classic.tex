% Custom commands - Enhanced with styling

% spacing command
\newcommand{\sectionSpacing}{\vspace{-9pt}}

% Single source of truth for horizontal alignment of all text/bullets.
% Every list indents by this amount on BOTH sides, and every tabular inside a
% list is sized to \textwidth - 2*\resumeMargin so left AND right edges line up.
\newlength{\resumeMargin}
\setlength{\resumeMargin}{0.1in}
% Gap between a bullet/label and its item text. Tables inside empty-label lists
% must subtract this too so their right edge lines up with the bullets' text.
\newlength{\resumeLabelSep}
\setlength{\resumeLabelSep}{0.5em}
% Vertical gap between bullets within a list. One knob to loosen/tighten.
\newlength{\resumeItemSep}
\setlength{\resumeItemSep}{3pt}

% Generic Item Building Functions (with color support)
\newcommand{\listStart}[1]{\begin{itemize}[labelindent=\resumeMargin, labelsep=\resumeLabelSep, leftmargin=*, rightmargin=\resumeMargin, align=left, itemsep=\resumeItemSep, parsep=0pt, label={#1}]}
\newcommand{\subListStart}[1]{\begin{itemize}[labelindent=-\resumeMargin, labelsep=\resumeLabelSep, leftmargin=*, rightmargin=\resumeMargin, align=left, itemsep=\resumeItemSep, parsep=0pt, label={#1}]}
\newcommand{\listItem}[1]{\item{\color{darkgray}{#1}}}
\newcommand{\listEnd}{\end{itemize}} \vspace{5pt}

% Specific Item Building Functions with Enhanced Styling
\newcommand{\resumeTechnicalSkillsListStart}{\listStart{\textcolor{successgreen}{\ding{51}}}}
\newcommand{\resumeKeyAchievementsListStart}{\listStart{\textcolor{mediumgray}{\textbullet}}}

% Enhanced resume subheading with better typography and spacing
\newcommand{\resumeSubheading}[4]{
  \vspace{-3pt} \item 
    \hspace{-\resumeLabelSep}\begin{tabular*}{\dimexpr\textwidth-2\resumeMargin\relax}[t]{l@{\extracolsep{\fill}}r}
      \textbf{\color{accentblue}#1} & \color{mediumgray}#2 \\
      \color{mediumgray}#3 & \color{mediumgray}#4 \\
    \end{tabular*}
  \vspace{-3pt}
}

\newcommand{\educationItem}[4]{
  \vspace{-3pt} \item
    \hspace{-\resumeLabelSep}\begin{tabular*}{\dimexpr\textwidth-2\resumeMargin\relax}[t]{l@{\extracolsep{\fill}}r}
