\newcommand{\leftPanelBackground}[1]{
  \AddToShipoutPictureBG{
    \begin{tikzpicture}[remember picture,overlay]
      \fill[leftpanelbg] 
        ([xshift=-.1in,yshift=0in]current page.south west) 
        rectangle 
        ([xshift=#1\paperwidth,yshift=0in]current page.north west);
    \end{tikzpicture}
  }
}
